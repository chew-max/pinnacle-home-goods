<script>
(function () {
  "use strict";


  /*
   * A successful payment is the only point
   * where the customer's local cart is cleared.
   */

  localStorage.removeItem(
    "jekyllCommerceCart"
  );


  const params =
    new URLSearchParams(
      window.location.search
    );


  const orderReference =
    params.get("order");


  const token =
    params.get("token");


  /*
   * Display the customer-facing order reference
   * when one was supplied by the payment return.
   */

  if (orderReference) {

    const wrapper =
      document.getElementById(
        "order-reference"
      );


    const value =
      document.getElementById(
        "order-reference-value"
      );


    value.textContent =
      orderReference;


    wrapper.style.display =
      "block";
  }


  /*
   * The status token is a capability credential.
   *
   * Do not display it on the page. It is used
   * only to construct the customer's private
   * order-status URL.
   */

  if (token) {

    const statusButton =
      document.getElementById(
        "view-order-status"
      );


    const statusUrl =
      new URL(
        "{{ '/order-status/' | relative_url }}",
        window.location.origin
      );


    statusUrl.searchParams.set(
      "token",
      token
    );


    statusButton.href =
      statusUrl.toString();


    statusButton.hidden =
      false;
  }

})();
</script>

<script>
(function () {
  "use strict";


  /*
   * IMPORTANT:
   *
   * Do NOT clear the cart on this page.
   * Failed or incomplete checkout must preserve
   * the customer's cart for another attempt.
   */


  const params =
    new URLSearchParams(
      window.location.search
    );


  const reason =
    params.get("reason");


  const message =
    document.getElementById(
      "checkout-failed-message"
    );


  /*
   * Keep customer-facing messaging intentionally
   * generic. Provider or internal error details
   * should never be exposed here.
   */

  switch (reason) {

    case "declined":

      message.textContent =
        "Your payment was declined. No order was completed. " +
        "Your cart has been preserved so you can review your " +
        "payment information and try again.";

      break;


    case "cancelled":

    case "canceled":

      message.textContent =
        "Checkout was canceled before payment was completed. " +
        "Your cart has been preserved so you can continue when ready.";

      break;


    case "error":

      message.textContent =
        "We couldn't confirm your payment at this time. " +
        "Your cart has been preserved. Before trying again, " +
        "check your email for an order confirmation.";

      break;


    default:

      message.textContent =
        "Your checkout did not complete. " +
        "Your cart has been preserved so you can try again.";

      break;

  }

})();
</script>
